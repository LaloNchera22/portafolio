Monetize your player base immediately. Add competitive wagering with minimal engineering overhead using our Typescript and Rust SDKs.

// Initialize Runinback Client
const client = new RuninbackClient(process.env.RUNINBACK_API_KEY);

// Setup Match Event Listener
client.on('match_completed', async (matchId) => {
    const results = await gameServer.getMatchResults(matchId);

    // Push results to the Trustless Oracle for automatic settlement
    await client.settleEscrow(matchId, {
        winnerId: results.winner_wallet,
        score: results.score
    });
});
